cmake_minimum_required(VERSION 3.10)
project(DT08 CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(include third_party)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra)   # cảnh báo là lỗi cần sửa, không commit code còn warning
  add_compile_options(-finput-charset=UTF-8 -fexec-charset=UTF-8)
elseif(MSVC)
  add_compile_options(/utf-8)
endif()
file(GLOB LIB_SRC src/*.cpp)
list(REMOVE_ITEM LIB_SRC ${CMAKE_SOURCE_DIR}/src/main.cpp ${CMAKE_SOURCE_DIR}/src/main_test.cpp ${CMAKE_SOURCE_DIR}/src/HttpApi.cpp)
add_executable(dt08 src/main.cpp src/HttpApi.cpp ${LIB_SRC})
add_executable(dt08_test src/main_test.cpp ${LIB_SRC})
find_package(Threads REQUIRED)
target_link_libraries(dt08 Threads::Threads)
if(WIN32)
  target_compile_definitions(dt08 PRIVATE _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN)
  target_link_libraries(dt08 ws2_32 crypt32)
endif()

include(CTest)
if(BUILD_TESTING)
  add_test(NAME dt08_test COMMAND dt08_test)
  set_tests_properties(dt08_test PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  include(CheckCXXCompilerFlag)
  check_cxx_compiler_flag("-fsanitize=address" DT08_HAS_ASAN)
  if(DT08_HAS_ASAN)
    target_compile_options(dt08_test PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(dt08_test PRIVATE -fsanitize=address)
  endif()
endif()
